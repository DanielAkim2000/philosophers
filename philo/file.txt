#include "../../includes/parsing.h"

static int	fill_rules(t_rules *rules, int ac, char **av)
{
	rules->nb_philo = ft_atoi(av[1]);
	if (rules->nb_philo < 0)
		return (1);
	rules->t_die = ft_atol(av[2]);
	if (rules->t_die < 0)
		return (1);
	rules->t_eat = ft_atol(av[3]);
	if (rules->t_eat < 0)
		return (1);
	rules->t_sleep = ft_atol(av[4]);
	if (rules->t_sleep < 0)
		return (1);
	if (ac == 6)
	{
		rules->nb_meals = ft_atoi(av[5]);
		if (rules->nb_meals < 0)
			return (1);
	}
	else
		rules->nb_meals = -1;
	rules->is_dead = 0;
	return (0);
}

t_rules	*ft_parse_arg(int ac, char **av)
{
	t_rules	*rules;

	if (ac != 5 && ac != 6)
		return (NULL);
	rules = malloc(sizeof(t_rules));
	if (!rules)
		return (NULL);
	if (fill_rules(rules, ac, av))
	{
		ft_free((void **)&rules);
		return (NULL);
	}
	return (rules);
}

int	ft_isdigit(char c)
{
	if (c < '0' || c > '9')
		return (0);
	return (1);
}

int	ft_atoi(char *str)
{
	int	i;
	int	mul;
	int	result;
	int	digit;

	i = 0;
	mul = -1;
	result = 0;
	while (str[i] == 32 || (str[i] >= '\t' && str[i] <= '\r'))
		i++;
	if (str[i] && (str[i] == '+' || str[i] == '-'))
		if (str[i++] == '-')
			mul = mul * (-1);
	while (str[i] && ft_isdigit(str[i]))
	{
		digit = str[i] - 48;
		if (result < (INT_MIN + digit) / 10)
			return (-1);
		result = result * 10 - (str[i++] - 48);
	}
	return (result * mul);
}

long	ft_atol(char *str)
{
	int		i;
	int		mul;
	long	result;
	int		digit;

	i = 0;
	mul = -1;
	result = 0;
	while (str[i] == 32 || (str[i] >= '\t' && str[i] <= '\r'))
		i++;
	if (str[i] && (str[i] == '+' || str[i] == '-'))
		if (str[i++] == '-')
			mul = mul * (-1);
	while (str[i] && ft_isdigit(str[i]))
	{
		digit = str[i] - 48;
		if (result < (LONG_MIN + digit) / 10)
			return (-1);
		result = result * 10 - (str[i++] - 48);
	}
	return (result * mul);
}

void	ft_free(void **addr_ptr)
{
	if (addr_ptr && *addr_ptr)
	{
		free(*addr_ptr);
		*addr_ptr = NULL;
	}
}

void	ft_safe_print(t_philo *p, char *msg)
{
	long	time;

	pthread_mutex_lock(&p->rules->print);
	pthread_mutex_lock(&p->rules->death);
	if (!p->rules->is_dead)
	{
		time = ft_gettime() - p->rules->start_time;
		printf("%ld %d %s\n", time, p->id, msg);
	}
	pthread_mutex_unlock(&p->rules->death);
	pthread_mutex_unlock(&p->rules->print);
}

long	ft_gettime(void)
{
	t_timeval	tv;
	long		time;

	if (gettimeofday(&tv, NULL) == -1)
		return (-1);
	time = tv.tv_sec * 1000 + tv.tv_usec / 1000;
	return (time);
}

void	ft_usleep(long t_ms, t_philo *p)
{
	long	start;

	start = ft_gettime();
	while (1)
	{
		pthread_mutex_lock(&p->rules->death);
		if (p->rules->is_dead)
		{
			pthread_mutex_unlock(&p->rules->death);
			return ;
		}
		pthread_mutex_unlock(&p->rules->death);
		if (ft_gettime() - start >= t_ms)
			break ;
		usleep(1000);
	}
}

static int	ft_check_death(t_philo *philos, int i, t_rules *rules)
{
	long	now;

	now = ft_gettime();
	pthread_mutex_lock(&rules->meal);
	if (now - philos[i].last_meal > rules->t_die)
	{
		pthread_mutex_lock(&rules->death);
		if (!rules->is_dead)
		{
			rules->is_dead = 1;
			pthread_mutex_unlock(&rules->death);
			pthread_mutex_lock(&rules->print);
			printf("%ld %d died\n", now - philos[i].rules->start_time,
				philos[i].id);
			pthread_mutex_unlock(&rules->print);
		}
		else
			pthread_mutex_unlock(&rules->death);
		pthread_mutex_unlock(&rules->meal);
		return (1);
	}
	pthread_mutex_unlock(&rules->meal);
	return (0);
}

void	ft_monitor(t_philo *philos)
{
	int		i;
	int		nb_full;
	t_rules	*rules;

	rules = philos[0].rules;
	while (1)
	{
		nb_full = 0;
		i = 0;
		while (i < rules->nb_philo)
		{
			if (ft_check_death(philos, i, rules))
				return ;
			pthread_mutex_lock(&rules->meal);
			if (rules->nb_meals > 0 && philos[i].is_full)
				nb_full++;
			pthread_mutex_unlock(&rules->meal);
			i++;
		}
		if (rules->nb_meals > 0 && nb_full == rules->nb_philo)
			return ;
		usleep(1000);
	}
}

void	*ft_routine_monitor(void *arg)
{
	t_philo	*p;

	p = (t_philo *)arg;
	ft_monitor(p);
	return (NULL);
}

#include "../includes/init.h"

int	ft_init_rules_mutex(t_rules *rules)
{
	if (pthread_mutex_init(&rules->print, NULL))
		return (1);
	if (pthread_mutex_init(&rules->death, NULL))
		return (pthread_mutex_destroy(&rules->print), 1);
	if (pthread_mutex_init(&rules->meal, NULL))
		return (pthread_mutex_destroy(&rules->print),
			pthread_mutex_destroy(&rules->death), 1);
	return (0);
}

pthread_mutex_t	*ft_init_forks(t_rules *rules)
{
	pthread_mutex_t	*forks;
	int				i;

	i = 0;
	forks = (pthread_mutex_t *)malloc(sizeof(pthread_mutex_t)
			* rules->nb_philo);
	if (!forks)
		return (NULL);
	while (i < rules->nb_philo)
	{
		if (pthread_mutex_init(&forks[i], NULL) != 0)
			return (ft_free((void **)&forks), NULL);
		i++;
	}
	return (forks);
}

t_philo	*ft_init_philos(t_rules *rules, pthread_mutex_t *forks)
{
	int		i;
	t_philo	*philos;
	int		right_index;

	i = -1;
	philos = (t_philo *)malloc(sizeof(t_philo) * rules->nb_philo);
	if (!philos)
		return (NULL);
	while (++i < rules->nb_philo)
	{
		philos[i].id = i + 1;
		philos[i].meals_eaten = 0;
		philos[i].rules = rules;
		philos[i].left_fork = &forks[i];
		right_index = (i + 1) % rules->nb_philo;
		philos[i].right_fork = &forks[right_index];
		philos[i].is_full = 0;
		philos[i].last_meal = 0;
	}
	return (philos);
}

static void	ft_routine_gap(t_philo *p)
{
	if (p->rules->nb_philo % 2 == 0)
	{
		if (p->id % 2 != 0)
			ft_usleep(p->rules->t_eat, p);
	}
	else
	{
		if (p->id == p->rules->nb_philo)
			ft_usleep(p->rules->t_eat * 2, p);
		else if (p->id % 2 != 0)
			ft_usleep(p->rules->t_eat, p);
	}
}

static void	ft_process_routine(t_philo *p)
{
	while (1)
	{
		pthread_mutex_lock(&p->rules->death);
		if (p->rules->is_dead)
		{
			pthread_mutex_unlock(&p->rules->death);
			break ;
		}
		pthread_mutex_unlock(&p->rules->death);
		pthread_mutex_lock(&p->rules->meal);
		if (p->is_full)
		{
			pthread_mutex_unlock(&p->rules->meal);
			break ;
		}
		pthread_mutex_unlock(&p->rules->meal);
		ft_take_forks(p);
		if (p->rules->nb_philo == 1)
			break ;
		ft_eat(p);
		ft_put_forks(p);
		ft_sleep_philo(p);
		ft_think(p);
	}
}

void	*ft_routine(void *arg)
{
	t_philo	*p;

	p = (t_philo *)arg;
	// pthread_mutex_lock(&p->rules->meal);
	// p->last_meal = ft_gettime();
	// pthread_mutex_unlock(&p->rules->meal);
	ft_routine_gap(p);
	ft_process_routine(p);
	return (NULL);
}

void	ft_process_philo_one(t_philo *p)
{
	pthread_mutex_lock(p->left_fork);
	ft_safe_print(p, "has taken a fork");
	ft_usleep(p->rules->t_die, p);
	pthread_mutex_unlock(p->left_fork);
}

void	ft_process_philo_peer(t_philo *p)
{
	pthread_mutex_lock(p->left_fork);
	ft_safe_print(p, "has taken a fork");
	pthread_mutex_lock(p->right_fork);
	ft_safe_print(p, "has taken a fork");
}

void	ft_process_philo_odd(t_philo *p)
{
	pthread_mutex_lock(p->right_fork);
	ft_safe_print(p, "has taken a fork");
	pthread_mutex_lock(p->left_fork);
	ft_safe_print(p, "has taken a fork");
}

void	ft_think(t_philo *p)
{
	long	think_time;
	long	time_margin;

	// pthread_mutex_lock(&p->rules->death);
	// if (p->rules->is_dead)
	// {
	// 	pthread_mutex_unlock(&p->rules->death);
	// 	return ;
	// }
	// pthread_mutex_unlock(&p->rules->death);
	ft_safe_print(p, "is thinking");
	if (p->rules->nb_philo % 2 != 0)
	{
		time_margin = 2 * p->rules->t_eat - p->rules->t_sleep;
		if (time_margin > 0)
			think_time = time_margin / 2;
		else
			think_time = 0;
		if (think_time > 0)
			ft_usleep(think_time, p);
	}
}

void	ft_take_forks(t_philo *p)
{
	if (p->rules->nb_philo == 1)
	{
		ft_process_philo_one(p);
		return ;
	}
	if (p->rules->nb_philo % 2 != 0 && p->id == p->rules->nb_philo)
	{
		pthread_mutex_lock(p->left_fork);
		ft_safe_print(p, "has taken a fork");
		pthread_mutex_lock(p->right_fork);
		ft_safe_print(p, "has taken a fork");
		return ;
	}
	if (p->id % 2 == 0)
	{
		pthread_mutex_lock(p->left_fork);
		ft_safe_print(p, "has taken a fork");
		pthread_mutex_lock(p->right_fork);
		ft_safe_print(p, "has taken a fork");
	}
	else
	{
		pthread_mutex_lock(p->right_fork);
		ft_safe_print(p, "has taken a fork");
		pthread_mutex_lock(p->left_fork);
		ft_safe_print(p, "has taken a fork");
	}
}

void	ft_put_forks(t_philo *p)
{
	pthread_mutex_unlock(p->left_fork);
	if (p->right_fork != p->left_fork)
		pthread_mutex_unlock(p->right_fork);
}

void	ft_eat(t_philo *p)
{
	ft_safe_print(p, "is eating");
	pthread_mutex_lock(&p->rules->meal);
	p->last_meal = ft_gettime();
	p->meals_eaten++;
	if (p->rules->nb_meals > 0 && p->meals_eaten >= p->rules->nb_meals)
		p->is_full = 1;
	pthread_mutex_unlock(&p->rules->meal);
	ft_usleep(p->rules->t_eat, p);
}

static int	ft_start_simulation(t_philo *philos)
{
	int		i;
	t_rules	*r;

	i = 0;
	r = philos->rules;
	while (i < philos[0].rules->nb_philo)
	{
		// philos[i].last_meal = r->start_time;
		if (pthread_create(&philos[i].thread, NULL, ft_routine, &philos[i]))
			return (i);
		i++;
	}
	if (pthread_create(&r->monitor, NULL, ft_routine_monitor, philos))
		return (-1);
	return (0);
}

void	ft_join_thread(t_rules *rules, t_philo *philos)
{
	int	i;

	i = -1;
	while (++i < rules->nb_philo)
		pthread_join(philos[i].thread, NULL);
	pthread_join(philos->rules->monitor, NULL);
}

void	ft_destroy(t_rules *rules, pthread_mutex_t *forks)
{
	int	i;

	i = -1;
	pthread_mutex_destroy(&rules->print);
	pthread_mutex_destroy(&rules->meal);
	pthread_mutex_destroy(&rules->death);
	if (forks)
	{
		while (++i < rules->nb_philo)
			pthread_mutex_destroy(&forks[i]);
	}
}

static void	ft_clean(t_rules *rules, pthread_mutex_t *forks, t_philo *p)
{
	ft_destroy(rules, forks);
	ft_free((void **)&forks);
	ft_free((void **)&p);
	ft_free((void **)&rules);
}

int	main(int ac, char **av)
{
	t_rules			*rules;
	t_philo			*philos;
	pthread_mutex_t	*forks;
	int				n;

	rules = ft_parse_arg(ac, av);
	if (!rules)
		return (1);
	if (ft_init_rules_mutex(rules))
		return (ft_free((void **)&rules), 1);
	forks = ft_init_forks(rules);
	if (!forks)
		return (ft_destroy(rules, NULL), ft_free((void **)&rules), 1);
	philos = ft_init_philos(rules, forks);
	if (!philos)
		return (ft_destroy(rules, forks), ft_free((void **)&rules),
			ft_free((void **)&forks), 1);
	rules->start_time = ft_gettime();
	n = ft_start_simulation(philos);
	if (!n)
		ft_join_thread(rules, philos);
	else
	{
		if (n == -1)
			pthread_join(rules->monitor, NULL);
		else
		{
			for (int i = 0; i < n; i++)
				pthread_join(philos[i].thread, NULL);
		}
	}
	ft_clean(rules, forks, philos);
}
